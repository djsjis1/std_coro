# Web 实现库: 根工程、独立 Web 入口、tests-only 共用同一编译单元。
if(TARGET coro::web)
    return()
endif()
if(NOT TARGET coro::coro)
    message(FATAL_ERROR "Configure the repository root or Web directory first")
endif()
get_target_property(CORO_WEB_IO_AVAILABLE coro CORO_WEB_IO_AVAILABLE)
if(NOT CORO_WEB_IO_AVAILABLE)
    message(FATAL_ERROR "Web requires Windows IOCP or Linux io_uring with file and network support")
endif()

if(NOT TARGET coro::http_protocol)
    message(FATAL_ERROR "The root project must configure the HTTP protocol component before Web")
endif()

add_library(coro_web STATIC http_types.cpp web_server.cpp)
add_library(coro::web ALIAS coro_web)
target_include_directories(coro_web PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(coro_web PUBLIC coro::io router::router PRIVATE coro::http_protocol)
if(MSVC)
    target_compile_options(coro_web PRIVATE /utf-8)
    set_property(TARGET coro_web PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
